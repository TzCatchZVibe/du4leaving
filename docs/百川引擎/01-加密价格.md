## §2 数学骨架 · Black-Scholes 二元定价

```
对一个 cash-or-nothing 看涨二元期权 ·
(Kalshi "BTC 在 $K 以上 by T" 的 yes 边)

公式 · P_yes = N(d2)

其中 ·
  d2 = [ ln(S/K) + (r - σ²/2) × T ] / (σ × √T)
  
  N()  · 标准正态累积分布函数
  S    · 当前现货 BTC 价
  K    · 行权价 (Kalshi 那个数)
  T    · 距到期年化时间 (天 / 365)
  σ    · 年化波动率
  r    · 无风险利率 (T-Bill 1 月 ≈ 0.045)
```

```
对二元 PUT (BTC 在 $K 以下) ·
  P_no = N(-d2) = 1 - N(d2)
```

```
对区间 (BTC 在 [K1, K2] 之间) ·
  P = N(d2_K1) - N(d2_K2)
```

```typescript
// src/lib/xiapan/options-pricing.ts
import { erf } from 'mathjs';

export function normalCDF(x: number): number {
  return 0.5 * (1 + erf(x / Math.sqrt(2)));
}

export interface BSInput {
  spot: number;
  strike: number;
  T_years: number;
  sigma_annual: number;
  r?: number;
}

export function binaryCallFairP(opts: BSInput): number {
  const { spot, strike, T_years, sigma_annual, r = 0.045 } = opts;
  if (T_years <= 0 || sigma_annual <= 0) return 0.5;
  const d2 = (Math.log(spot / strike) + (r - 0.5 * sigma_annual ** 2) * T_years)
             / (sigma_annual * Math.sqrt(T_years));
  return normalCDF(d2);
}

export function binaryPutFairP(opts: BSInput): number {
  return 1 - binaryCallFairP(opts);
}

export function rangeFairP(spot: number, K1: number, K2: number, T: number, σ: number, r = 0.045): number {
  // P(K1 < S_T < K2) = N(d2_K1) - N(d2_K2)
  if (K1 >= K2) [K1, K2] = [K2, K1];
  return binaryCallFairP({ spot, strike: K1, T_years: T, sigma_annual: σ, r }) -
         binaryCallFairP({ spot, strike: K2, T_years: T, sigma_annual: σ, r });
}

// realized-vol.ts
export function realizedVol(prices: number[], periodsPerYear = 365): number {
  if (prices.length < 10) return 0.6; // 默认 60%
  const returns: number[] = [];
  for (let i = 1; i < prices.length; i++) {
    returns.push(Math.log(prices[i] / prices[i - 1]));
  }
  const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
  const variance = returns.reduce((s, r) => s + (r - mean) ** 2, 0) / (returns.length - 1);
  return Math.sqrt(variance * periodsPerYear);
}
```

```typescript
// src/app/api/xiapan/btc-edges/route.ts (skeleton)
export async function GET() {
  // 1. 拉 Coinbase BTC 现货
  const spot = await fetchCoinbaseSpot('BTC-USD');
  
  // 2. 算 30d 已实现 vol
  const ohlc30d = await fetchCoinbaseOHLC('BTC-USD', '1d', 30);
  const sigma = realizedVol(ohlc30d.map(c => c.close)) * 1.05;
  
  // 3. 拉 Kalshi BTC 二元市场
  const markets = await fetchKalshiMarkets({ event_prefix: 'KXBTCD' });
  
  // 4. 对每个市场 · 解析 strike / T · 算 fairP
  const edges = markets.map(m => {
    const { strike, T_years } = parseKxbtcd(m.ticker);
    const fairP = binaryCallFairP({ spot, strike, T_years, sigma_annual: sigma });
    const marketP = m.yes_ask / 100;
    return {
      ticker: m.ticker,
      spot, strike, T_years, sigma,
      fairP, marketP,
      edge_pp: (fairP - marketP) * 100,
      vol_24: m.volume_24h,
      spread: m.yes_ask - m.yes_bid,
    };
  });
  
  // 5. 排序 + 返回
  edges.sort((a, b) => Math.abs(b.edge_pp) - Math.abs(a.edge_pp));
  return NextResponse.json({ ok: true, edges });
}
```
