## §2 Bayesian 融合

```typescript
// src/lib/xiapan/百川/fusion.ts

export interface FusionInput {
  ticker: string;
  market_implied_p: number;   // Kalshi yes_ask / 100
  signals: Signal[];           // 该 ticker 收到的所有信号
  signal_weights: Record<string, number>;  // source → 当前权重 (Brier 调)
}

export function fuse(input: FusionInput): {
  p_consensus: number;
  edge_pp: number;
  side: 1 | -1;
  n_active: number;
  kelly_full: number;
} {
  const { market_implied_p, signals, signal_weights } = input;
  
  // log-odds prior
  const prior = market_implied_p;
  let log_odds = Math.log(prior / (1 - prior));
  
  let n_active = 0;
  const directions = new Set<number>();
  
  for (const s of signals) {
    const w = signal_weights[s.source] ?? 1.0;
    const a = Math.max(0.51, Math.min(0.85, s.confidence));  // clip
    const contribution = s.direction * w * Math.log(a / (1 - a));
    log_odds += contribution;
    n_active++;
    directions.add(s.direction);
  }
  
  const p_consensus = 1 / (1 + Math.exp(-log_odds));
  const edge_pp = (p_consensus - market_implied_p) * 100;
  const side = edge_pp >= 0 ? 1 : -1;
  
  // 信号方向冲突 → edge 失效
  if (directions.size > 1) {
    return { p_consensus, edge_pp: 0, side, n_active, kelly_full: 0 };
  }
  
  const p_correct = side === 1 ? p_consensus : (1 - p_consensus);
  const cost = side === 1 ? market_implied_p : (1 - market_implied_p);
  const kelly_full = cost > 0 && cost < 1 ? (p_correct - cost) / (1 - cost) : 0;
  
  return { p_consensus, edge_pp, side, n_active, kelly_full };
}
```

## §3 决策门 + 仓位

```typescript
// src/lib/xiapan/百川/decide.ts

export interface DecideInput {
  fusion_result: ReturnType<typeof fuse>;
  vol_24: number;
  spread_c: number;
  bankroll: number;
  current_drawdown_pct: number;  // 0-1 · 当前 vs 峰值
}

export function decide(input: DecideInput): {
  act: boolean;
  reason: string;
  stake_usd: number;
  qty: number;
} {
  const { fusion_result: f, vol_24, spread_c, bankroll, current_drawdown_pct } = input;
  
  // 5 条门
  if (Math.abs(f.edge_pp) < 5) return { act: false, reason: 'edge < 5pp', stake_usd: 0, qty: 0 };
  if (f.p_consensus < 0.05 || f.p_consensus > 0.95) return { act: false, reason: 'tail event', stake_usd: 0, qty: 0 };
  if (vol_24 < 500) return { act: false, reason: 'low liquidity', stake_usd: 0, qty: 0 };
  if (spread_c > 3) return { act: false, reason: 'wide spread', stake_usd: 0, qty: 0 };
  if (f.n_active < 2) return { act: false, reason: 'single signal', stake_usd: 0, qty: 0 };
  if (current_drawdown_pct > 0.30) return { act: false, reason: 'deep drawdown · pause', stake_usd: 0, qty: 0 };
  
  // Kelly 多重缩
  const shrink_thorp = 0.25;
  const shrink_signals = Math.min(f.n_active / 3, 1) ** 2;
  const shrink_drawdown = (1 - current_drawdown_pct) ** 2;
  const shrink_edge = Math.min(Math.abs(f.edge_pp) / 15, 1);
  
  const kelly_use = f.kelly_full * shrink_thorp * shrink_signals * shrink_drawdown * shrink_edge;
  
  // 仓位约束 · 单笔 ≤ 2% · 最少 $1
  const stake_kelly = bankroll * kelly_use;
  const stake_capped = Math.min(stake_kelly, bankroll * 0.02);
  const stake_usd = Math.max(0, stake_capped);
  
  if (stake_usd < 1) return { act: false, reason: 'stake too small', stake_usd: 0, qty: 0 };
  
  return { act: true, reason: 'pass all gates', stake_usd, qty: 0 };  // qty 由 caller 计算
}
```

## §4 风控全栈

```typescript
// src/lib/xiapan/百川/risk.ts

export const RISK_LIMITS = {
  single_trade_max_pct: 0.02,       // 2% bankroll
  daily_exposure_max_pct: 0.06,     // 6%
  correlated_max_pct: 0.04,         // 4% · 同 event 多 market
  consecutive_loss_pause_count: 5,   // 连输 5 单暂停
  consecutive_loss_pause_hours: 6,
  daily_loss_circuit_pct: 0.08,     // -8% 日 · 24h 暂停
  weekly_loss_circuit_pct: 0.15,    // -15% 周 · 7d 暂停
  min_stake_usd: 1,
  max_stake_usd: 10,
  min_score: 5,                      // edge_pp ≥ 5
  max_score: 30,                     // edge_pp > 30 警告 · 多半幻觉
};

export interface RiskCheckInput {
  proposed_stake: number;
  bankroll: number;
  today_open_exposure: number;
  today_pnl: number;
  week_pnl: number;
  consecutive_losses: number;
  last_trade_ts?: string;
  ticker: string;
  ticker_event?: string;
  open_event_exposure: number;
}

export function riskCheck(input: RiskCheckInput): {
  pass: boolean;
  reason: string;
  effective_stake: number;
} {
  const { 
    proposed_stake, bankroll, today_open_exposure, today_pnl, week_pnl,
    consecutive_losses, open_event_exposure
  } = input;
  
  // 单笔
  const single_max = bankroll * RISK_LIMITS.single_trade_max_pct;
  if (proposed_stake > single_max) {
    return { pass: false, reason: 'single trade > 2%', effective_stake: single_max };
  }
  
  // 日 exposure
  const daily_max = bankroll * RISK_LIMITS.daily_exposure_max_pct;
  if (today_open_exposure + proposed_stake > daily_max) {
    return { pass: false, reason: 'daily exposure cap', effective_stake: 0 };
  }
  
  // 关联
  const corr_max = bankroll * RISK_LIMITS.correlated_max_pct;
  if (open_event_exposure + proposed_stake > corr_max) {
    return { pass: false, reason: 'correlated cap', effective_stake: 0 };
  }
  
  // 连续亏
  if (consecutive_losses >= RISK_LIMITS.consecutive_loss_pause_count) {
    return { pass: false, reason: 'consecutive loss pause', effective_stake: 0 };
  }
  
  // 日熔断
  if (today_pnl < -bankroll * RISK_LIMITS.daily_loss_circuit_pct) {
    return { pass: false, reason: '24h circuit · -8%', effective_stake: 0 };
  }
  
  // 周熔断
  if (week_pnl < -bankroll * RISK_LIMITS.weekly_loss_circuit_pct) {
    return { pass: false, reason: '7d circuit · -15%', effective_stake: 0 };
  }
  
  return { pass: true, reason: 'pass', effective_stake: proposed_stake };
}
```

## §5 反馈环 · Brier 自适应

```typescript
// src/lib/xiapan/百川/calibrate.ts

export interface ClosedTrade {
  source: string;            // 触发的 signal source
  predicted_p: number;
  actual: 0 | 1;             // 平仓 yes 中 = 1
  pnl: number;
  ts_open: string;
  ts_close: string;
}

export function brierBySource(trades: ClosedTrade[]): Record<string, number> {
  const groups = new Map<string, ClosedTrade[]>();
  for (const t of trades) {
    const arr = groups.get(t.source) ?? [];
    arr.push(t);
    groups.set(t.source, arr);
  }
  
  const result: Record<string, number> = {};
  for (const [src, ts] of groups) {
    if (ts.length < 10) continue;  // 样本不够
    const brier = ts.reduce((s, t) => s + (t.predicted_p - t.actual) ** 2, 0) / ts.length;
    result[src] = brier;
  }
  return result;
}

export function updateWeights(
  current: Record<string, number>,
  brier: Record<string, number>
): Record<string, number> {
  const updated: Record<string, number> = {};
  for (const src in brier) {
    const b = brier[src];
    // 反 Brier · 良 Brier (≤0.20) → 权重 1.5 / 差 (≥0.30) → 0.5
    if (b <= 0.18) updated[src] = 1.5;
    else if (b <= 0.22) updated[src] = 1.2;
    else if (b <= 0.27) updated[src] = 1.0;
    else if (b <= 0.32) updated[src] = 0.7;
    else if (b <= 0.40) updated[src] = 0.4;
    else updated[src] = 0.0;        // 退役
  }
  return updated;
}
```

## §6 Deflated Sharpe (退役坏 source)

```typescript
// 当 trades ≥ 200 后跑

export function deflatedSharpeRatio(returns: number[], n_strategies: number): number {
  if (returns.length < 30) return 0;
  const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
  const std = Math.sqrt(returns.reduce((s, r) => s + (r - mean) ** 2, 0) / (returns.length - 1));
  if (std === 0) return 0;
  
  const sr = mean / std * Math.sqrt(252);  // annualized
  
  // de Prado deflation factor
  const epsilon = 0.001;
  const deflated = sr / Math.sqrt(1 + epsilon * n_strategies);
  return deflated;
}
```

## §7 主入口 · cron 每分钟跑

```typescript
// src/app/api/xiapan/百川/run/route.ts

import { allProducers } from '@/lib/xiapan/百川/producers';

export async function GET() {
  // 1. 拉所有信号 (各品类并行)
  const all_signals = (await Promise.all(
    allProducers.map(p => p.produce().catch(() => []))
  )).flat();
  
  // 2. 按 ticker 聚合
  const by_ticker = new Map<string, Signal[]>();
  for (const s of all_signals) {
    const arr = by_ticker.get(s.ticker) ?? [];
    arr.push(s);
    by_ticker.set(s.ticker, arr);
  }
  
  // 3. 拉 source 当前权重 (持久化在 ~/.du4leaving/百川/weights.json)
  const weights = loadWeights();
  
  // 4. 对每 ticker 走 fusion + decide + risk
  const decisions: any[] = [];
  for (const [ticker, signals] of by_ticker) {
    if (signals.length < 2) continue;
    
    const market = await fetchKalshiMarket(ticker);
    const fusion_result = fuse({
      ticker,
      market_implied_p: market.yes_ask / 100,
      signals,
      signal_weights: weights,
    });
    
    const decision = decide({
      fusion_result,
      vol_24: market.volume_24h,
      spread_c: market.yes_ask - market.yes_bid,
      bankroll: getBankroll(),
      current_drawdown_pct: getCurrentDrawdown(),
    });
    
    if (!decision.act) continue;
    
    const risk = riskCheck({...});
    if (!risk.pass) continue;
    
    decisions.push({ ticker, decision, risk, fusion_result });
  }
  
  // 5. 落 paper trade · 写 lessons.jsonl
  for (const d of decisions) {
    await placePaperTrade(d);
  }
  
  return NextResponse.json({ ok: true, decisions });
}
```
